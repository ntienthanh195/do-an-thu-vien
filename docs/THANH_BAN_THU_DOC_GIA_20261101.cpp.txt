const int MUON_DANG_MUON = 0;
const int THE_HOAT_DONG = 1;

struct NgayThang
{
    int ngay = 0;
    int thang = 0;
    int nam = 0;
};

struct MuonTra
{
    string maSach;
    NgayThang ngayMuon;
    NgayThang ngayTra;
    int trangThai = MUON_DANG_MUON;
    MuonTra* next = nullptr;
};

struct DocGia
{
    int maThe = 0;
    string ho;
    string ten;
    string phai;
    int trangThaiThe = THE_HOAT_DONG;

    MuonTra* dsMuonTra = nullptr;
};

struct NutDocGia
{
    DocGia thongTin;
    NutDocGia* left = nullptr;
    NutDocGia* right = nullptr;
};

struct CayDocGia
{
    NutDocGia* root = nullptr;
};

NutDocGia* tim_cay(CayDocGia &caydocgia, int x)
{
    NutDocGia* p = caydocgia.root;

    while (p != nullptr)
    {
        if (x < p->thongTin.maThe)
            p = p->left;
        else if (x > p->thongTin.maThe)
            p = p->right;
        else if (x == p->thongTin.maThe)
            return p;
    }

    return nullptr;
}

void them_cay(CayDocGia &caydocgia, const DocGia &docGiaMoi)
{
    NutDocGia* cha = nullptr;
    NutDocGia* p = caydocgia.root;

    while (p != nullptr)
    {
        cha = p;

        if (docGiaMoi.maThe < p->thongTin.maThe)
            p = p->left;
        else if (docGiaMoi.maThe > p->thongTin.maThe)
            p = p->right;
        else if (docGiaMoi.maThe == p->thongTin.maThe)
        {
            cout << "Da co san";
            return;
        }
    }

    p = new NutDocGia;
    p->thongTin = docGiaMoi;

    if (cha == nullptr)
        caydocgia.root = p;
    else if (docGiaMoi.maThe < cha->thongTin.maThe)
        cha->left = p;
    else
        cha->right = p;
}

void them_lich_su(NutDocGia &q, MuonTra muontra)
{
	MuonTra* muontraa=new MuonTra;
	*muontraa= muontra;
	 muontraa->next = nullptr;  
	MuonTra* p=q.thongTin.dsMuonTra;
	while(p!=nullptr && p->next!=nullptr)
	{
		p=p->next;
	
	}
	if(p==nullptr)
	{
		p=muontraa;
		q.thongTin.dsMuonTra=p;
	}
	else 
	{
		p->next=muontraa;
	}
}

int Dem_so_sach_dang_muon(NutDocGia &q)
{
	int sachdangmuon=0;
	MuonTra* p=q.thongTin.dsMuonTra;
	while(p!=nullptr)
	{
		if(p->trangThai==0)
		sachdangmuon++;
		p=p->next;
	}
	return sachdangmuon;
}

MuonTra* TimMuonDangMuonTheoMa(NutDocGia &q,string maSach)
{
	
	MuonTra* p=q.thongTin.dsMuonTra;
	while(p!=nullptr)
	{
		if(p->maSach==maSach&&p->trangThai==0)
		{
			return p;
		}
		p=p->next;	
	}
	return nullptr;
	
}

bool NamNhuan(int nam)
{
    return nam % 400 == 0 ||
           (nam % 4 == 0 && nam % 100 != 0);
}


int SoNgayTrongThang(int thang, int nam)
{
    if (thang < 1 || thang > 12 || nam <= 0)
        return 0;

    switch (thang)
    {
        case 1:
        case 3:
        case 5:
        case 7:
        case 8:
        case 10:
        case 12:
            return 31;

        case 4:
        case 6:
        case 9:
        case 11:
            return 30;

        case 2:
            if (NamNhuan(nam))
                return 29;
            return 28;
    }

    return 0;
}

bool NgayHopLe(const NgayThang &ngay)
{
    int soNgay = SoNgayTrongThang(ngay.thang, ngay.nam);

    return soNgay > 0 &&
           ngay.ngay >= 1 &&
           ngay.ngay <= soNgay;
}

long long Doirangay(NgayThang sngay)
{
	long long soNgay=0;
	for(int i=1;i<sngay.nam;i++)
	{
		if(NamNhuan(i))
		{
			soNgay+= 366;
		}
		else
		{
			soNgay+= 365;
		}
	}
	

	for(int i=1;i<sngay.thang;i++)
	{
		soNgay+=SoNgayTrongThang(i,sngay.nam);
	}
	soNgay = soNgay + sngay.ngay-1;
	return soNgay;
}

long long KhoangCachNgay(NgayThang ngayt1, NgayThang ngayt2)
{
    long long i=Doirangay(ngayt1);
	long long j=Doirangay(ngayt2);
	return j-i;
}


bool CoSachQuaHan7Ngay(const NutDocGia &docgia, NgayThang ngayktra)
{
	MuonTra* q = docgia.thongTin.dsMuonTra;
	while(q!=nullptr)
	{
		if(q->trangThai == 0 )
		{
			long long k=KhoangCachNgay(q->ngayMuon,ngayktra);
			if(k>=7) return true; 
		}
		q=q->next;
	}
	return false;
}


bool CapNhatLuotTra(const NutDocGia &docgia,string masach, NgayThang ngaytrasach )
{
	MuonTra* q = docgia.thongTin.dsMuonTra;
	while(q!=nullptr && (q->maSach!=masach || q->trangThai!=0 ))
	{	
		q=q->next;
	}
	if(q==nullptr)
	{
		return false;
	}
	q->ngayTra=ngaytrasach;
	q->trangThai=1;
	return true;
}

void HuyDanhSachMuonTra(NutDocGia &docgia)
{
    MuonTra* q = docgia.thongTin.dsMuonTra;

    while (q != nullptr)
    {
        MuonTra* p = q;
        q = q->next;
        delete p;
    }

    docgia.thongTin.dsMuonTra = nullptr;
}




void Huydocgia(NutDocGia* &nutdocgia)
{
    if (nutdocgia != nullptr)
    {
        
        Huydocgia(nutdocgia->left);
        Huydocgia(nutdocgia->right);

        
        HuyDanhSachMuonTra(*nutdocgia);

        
        delete nutdocgia;
        nutdocgia = nullptr;
    }
}


void HuyCayDocGia(CayDocGia &caydocgia)
{
    Huydocgia(caydocgia.root);
}
